&НаСервере
Процедура ЗагрузитьКурсыНаСервере()
	
	Сервер = "www.cbr-xml-daily.ru"; 
	АдресРесурса = "www.cbr-xml-daily.ru/daily_json.js";
	Защита = Новый ЗащищенноеСоединениеOpenSSL(Неопределено, Новый СертификатыУдостоверяющихЦентровWindows());
	HTTPСоединение = Новый HTTPСоединение(Сервер, 443, , , , , Защита); 
	Запрос = Новый HTTPЗапрос("/daily_json.js");
	
	Попытка
		Ответ = HTTPСоединение.Получить(Запрос);
		ТекстОтвета = Ответ.ПолучитьТелоКакСтроку();
	Исключение
		Сообщить("Ошибка подключения к интернету: " + ОписаниеОшибки());
		Возврат;
	КонецПопытки;
	
	ЧтениеJSON = Новый ЧтениеJSON();
	Попытка 
		ЧтениеJSON.УстановитьСтроку(ТекстОтвета);
		Данные = ПрочитатьJSON(ЧтениеJSON);
	Исключение
		Сообщить("Ошибка: Сервер прислал данные в неверном формате (не JSON).")
	КонецПопытки;
	
	ЧтениеJSON.Закрыть();   
	
	Если Данные <> Неопределено И Данные.Свойство("Valute") Тогда
		Данные.Свойство("Valute"); 
		Доллар = Данные.Valute.USD;
		Евро = Данные.Valute.EUR;
		
		Сообщить("Курс доллара: " + Доллар.Value + " руб.");
		Сообщить("Курс евро: " + Евро.Value + " руб.");
		
		ТаблицаРезультат.Очистить(); 
		
		СтрокаДоллар = ТаблицаРезультат.Добавить();
		СтрокаДоллар.Валюта = "Доллар США";
		СтрокаДоллар.Курс = Доллар.Value;
		
		СтрокаЕвро = ТаблицаРезультат.Добавить();
		СтрокаЕвро.Валюта = "Евро";
		СтрокаЕвро.Курс = Евро.Value;
	Иначе
		Если Данные <> Неопределено Тогда
			Сообщить("Не удалось найти данные о валютах в ответе.");   
		КонецЕсли;
	КонецЕсли;
